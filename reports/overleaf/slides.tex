% 6. Why controlled ablation
\begin{frame}{Vì sao cần controlled ablation?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{block}{So sánh P3--P2}
        P3 và P2 được huấn luyện riêng, nên chênh lệch giữa hai mô hình có thể đến từ nhiều yếu tố khác nhau.

        \vspace{0.5em}
        So sánh P3--P2 không thể cô lập riêng ảnh hưởng của prepared context.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Cách kiểm tra}
        Để kiểm tra ảnh hưởng của context, ta giữ nguyên checkpoint P3 và toàn bộ điều kiện rollout, chỉ thay phần prepared context.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
